import axios from 'axios'; import * as cheerio from 'cheerio'; export default async function handler(req, res) { // Permitir CORS res.setHeader('Access-Control-Allow-Origin', '*'); res.setHeader('Access-Control-Allow-Methods', 'GET, POST, OPTIONS'); res.setHeader('Access-Control-Allow-Headers', 'Content-Type'); if (req.method === 'OPTIONS') { return res.status(200).end(); } if (req.method !== 'GET' && req.method !== 'POST') { return res.status(405).json({ error: 'Method not allowed' }); } try { const tweetUrl = req.query.url || req.body.url; if (!tweetUrl) { return res.status(400).json({ error: 'URL do tweet é necessária' }); } // Extrair ID do tweet const tweetIdMatch = tweetUrl.match(/\/status\/(\d+)/); if (!tweetIdMatch) { return res.status(400).json({ error: 'URL do tweet inválida' }); } const tweetId = tweetIdMatch[1]; // Tentar método 1: API do TwitSave try { const response = await axios.get(`https://twitsave.com/info?url=${encodeURIComponent(tweetUrl)}`, { headers: { 'User-Agent': 'Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36' } }); if (response.data.data && response.data.data.videos) { const videos = response.data.data.videos.map((video, idx) => ({ url: video.url, quality: video.quality || `Versão ${idx + 1}` })); return res.status(200).json({ videos, source: 'twitsave' }); } } catch (e) { console.log('TwitSave failed, trying alternative...'); } // Tentar método 2: Fazer scraping direto do HTML da página try { const pageResponse = await axios.get(`https://twitter.com/i/web/status/${tweetId}`, { headers: { 'User-Agent': 'Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36', 'Accept': 'text/html,application/xhtml+xml,application/xml;q=0.9,*/*;q=0.8' } }); const $ = cheerio.load(pageResponse.data); const videoUrls = []; // Procurar por URLs de vídeo no HTML $('video').each((i, el) => { const src = $(el).attr('src'); if (src) videoUrls.push(src); }); $('source').each((i, el) => { const src = $(el).attr('src'); if (src && src.includes('video')) videoUrls.push(src); }); if (videoUrls.length > 0) { const videos = videoUrls.map((url, idx) => ({ url: url, quality: `Versão ${idx + 1}` })); return res.status(200).json({ videos, source: 'direct' }); } } catch (e) { console.log('Direct scraping failed'); } return res.status(404).json({ error: 'Nenhum vídeo encontrado. Certifique-se de que o tweet é público e contém um vídeo.' }); } catch (error) { console.error('Error:', error); return res.status(500).json({ error: 'Erro ao processar o tweet. Tente novamente.' }); } }
